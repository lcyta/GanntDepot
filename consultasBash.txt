*- Consultar los 20 primeros Radon 

radon cc app/ --min A -s | ForEach-Object {
    if ($_ -match '\((\d+)\)$') {
        $score = [int]$matches[1]
        [pscustomobject]@{
            Line     = $_
            Score    = $score
        }
    }
} | Sort-Object -Property Score -Descending | Select-Object -First 20 | ForEach-Object {
    $_.Line
}



////////////////////////////////


lista de scrtips con mas lineas

Get-ChildItem -Recurse -Include *.py -Path app | ForEach-Object {
    $lineCount = (Get-Content $_.FullName).Count
    if ($lineCount -gt 50) {
        [PSCustomObject]@{
            File       = $_.FullName
            LineCount  = $lineCount
        }
    }
} | Sort-Object LineCount -Descending | Format-Table -AutoSize
